    /* Fare Calculation */
    function calculateDistance(lat1, lon1, lat2, lon2) {
      const R = 6371;
      const dLat = (lat2 - lat1) * Math.PI / 180;
      const dLon = (lon2 - lon1) * Math.PI / 180;
      const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
                Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
                Math.sin(dLon/2) * Math.sin(dLon/2);
      return R * (2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)));
    }

    function updateCalculatedPrices() {
      const dist = calculateDistance(
        activeRide.pickup.lat, activeRide.pickup.lng,
        activeRide.dropoff.lat, activeRide.dropoff.lng

      activeRide.distanceKm = parseFloat(dist.toFixed(1));

      Object.keys(VEHICLE_RATES).forEach(type => {
        const rate = VEHICLE_RATES[type];
        const price = Math.round(rate.base + (dist * rate.perKm));
        const el = document.getElementById(`price-${type}`);
        if (el) el.innerText = `PKR ${price}`;
        
        if (type === selectedVehicle) {
          activeRide.fare = price;
          document.getElementById('bookRidePriceTag').innerText = `PKR ${price}`;
        }
      });
    }

        selectedEl.classList.add('border-brand-500', 'bg-brand-500/10');
      }

      updateCalculatedPrices();
    }

    function updateDriverStepUI() {

      document.getElementById('driverRideStatusText').innerText = step.label;
      document.getElementById('driverStepBtnText').innerText = step.btnText;

      document.getElementById('driverPickupLabel').innerText = activeRide.pickup.nameUr;
      document.getElementById('driverDropoffLabel').innerText = activeRide.dropoff.nameUr;
      document.getElementById('driverFareEst').innerText = `PKR ${activeRide.fare}`;
      document.getElementById('driverDistanceEst').innerText = `${activeRide.distanceKm} km`;

      const stepBtn = document.getElementById('driverStepBtn');
      if (activeRide.driverStepIndex > 0) {
        stepBtn.disabled = false;
        stepBtn.className = "w-full bg-brand-600 hover:bg-brand-500 text-white font-bold py-3 px-4 rounded-xl transition flex items-center justify-center gap-2 text-xs shadow-lg shadow-brand-600/30 cursor-pointer";
      } else {
        stepBtn.disabled = true;
        stepBtn.className = "w-full bg-slate-700 text-slate-400 font-bold py-3 px-4 rounded-xl transition flex items-center justify-center gap-2 text-xs";
      }
    }

    function simulateDriverStep() {
      activeRide.driverStepIndex++;

      if (activeRide.driverStepIndex === 2) {
        document.getElementById('simBannerText').innerText = "کپتان پک اپ لوکیشن پر پہنچ چکے ہیں";
        driverMarker.setLatLng([activeRide.pickup.lat, activeRide.pickup.lng]);
      } else if (activeRide.driverStepIndex === 3) {
        document.getElementById('simBannerText').innerText = "سفر جاری ہے...";
      } else if (activeRide.driverStepIndex === 4) {
        document.getElementById('simBannerText').innerText = "سفر مکمل ہو چکا ہے!";
        driverMarker.setLatLng([activeRide.dropoff.lat, activeRide.dropoff.lng]);

        setTimeout(() => {
          document.getElementById('simulationBanner').classList.add('hidden');
          document.getElementById('ratingModal').classList.remove('hidden');
          activeRide.driverStepIndex = 0;
          updateDriverStepUI();
        }, 1000);
      }

      updateDriverStepUI();
    }

    function toggleDriverStatus() {
      const isChecked = document.getElementById('driverOnlineToggle').checked;
      document.getElementById('driverStatusLabel').innerText = isChecked ? "آن لائن" : "آف لائن";
    }

    function closeRatingModal() {
      document.getElementById('ratingModal').classList.add('hidden');
      switchMode('passenger');
    }

    /* Mode & Language Switchers */
    function switchMode(mode) {
      currentMode = mode;
      ['Passenger', 'Driver', 'Admin'].forEach(m => {
        const btn = document.getElementById(`modeBtn${m}`);
        if (m.toLowerCase() === mode) {
          btn.className = "px-3 py-1.5 rounded-lg transition flex items-center gap-1.5 bg-brand-600 text-white shadow-sm";
        } else {
          btn.className = "px-3 py-1.5 rounded-lg transition flex items-center gap-1.5 text-slate-400 hover:text-slate-200";
        }
      });

      document.getElementById('passengerSheet').classList.add('hidden');
      document.getElementById('driverSheet').classList.add('hidden');
      document.getElementById('adminSheet').classList.add('hidden');

      if (mode === 'passenger') document.getElementById('passengerSheet').classList.remove('hidden');
      else if (mode === 'driver') document.getElementById('driverSheet').classList.remove('hidden');
      else if (mode === 'admin') document.getElementById('adminSheet').classList.remove('hidden');

      setTimeout(() => { if (map) map.invalidateSize(); }, 200);
    }

    function toggleLanguage() {
      currentLang = currentLang === 'ur' ? 'en' : 'ur';
      document.getElementById('langLabel').innerText = currentLang === 'ur' ? 'English' : 'اردو';

      if (currentLang === 'en') {
        document.documentElement.setAttribute('dir', 'ltr');
        document.querySelectorAll('.lang-ur').forEach(el => el.classList.add('hidden'));
        document.querySelectorAll('.lang-en').forEach(el => el.classList.remove('hidden'));
      } else {
        document.documentElement.setAttribute('dir', 'rtl');
        document.querySelectorAll('.lang-en').forEach(el => el.classList.add('hidden'));
        document.querySelectorAll('.lang-ur').forEach(el => el.classList.remove('hidden'));
      }
    }
